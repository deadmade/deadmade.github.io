{{- /* One drawing's run layer, from data/runs.toml: an outline per stop and span, then a dot per hop. Hidden in base.css; motion.css shows it and times it. Called with (dict "name" "o" "h" or "v"). */ -}} {{- $d := index hugo.Data.runs.drawing .name -}} {{- if not $d }}{{ errorf "run.html: data/runs.toml has no drawing %q" .name }}{{ end -}} {{- $o := .o -}} {{- $outlines := slice -}} {{- range $d.stop }}{{ $outlines = $outlines | append (dict "cls" (printf "hl-%s" .id) "shape" (index . $o)) }}{{ end -}} {{- range $i, $s := $d.span }}{{ $outlines = $outlines | append (dict "cls" (printf "hl-span%d" $i) "shape" (index $s $o)) }}{{ end }} {{- range $outlines }}{{ $s := .shape }} {{- with $s.rect }}{{ end }} {{- with $s.polygon }}{{ end }} {{- with $s.path }}{{ end }} {{- with $s.text }}{{ index . 2 }}{{ end -}} {{- end }} {{- $n := 0 }} {{- range $d.run }}{{ range .hops }} {{- $dd := "" }} {{- range $i, $p := index . $o }}{{ $c := "L" }}{{ if eq $i 0 }}{{ $c = "M" }}{{ end }}{{ $dd = printf "%s%s%v %v" $dd $c (index $p 0) (index $p 1) }}{{ end }} {{- $n = add $n 1 }} {{- end }}{{ end }}
